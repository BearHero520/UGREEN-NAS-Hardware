"""Trace ugpt's actual plugin branch without importing or running vendor code."""
from pathlib import Path
import ast, hashlib, json
ROOT=Path(__file__).resolve().parents[4]
OUT=Path(__file__).resolve().parent
path=OUT/'payload/plugin/upgrade_tool.py'
source=path.read_text(encoding='utf-8'); tree=ast.parse(source)
funcs=[node for node in ast.walk(tree) if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef))]
names={node.name:node for node in funcs}
get=lambda name:ast.get_source_segment(source,names[name])
mcu=names['updata_mcu_fireware']; main=next(node for node in ast.walk(next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name=='Tool')) if isinstance(node,ast.FunctionDef) and node.name=='main')
checks=[]
def check(name,condition):
    assert condition,name
    checks.append({'name':name,'passed':True})
check('MCU source restricted to CEC-named wildcard','UGREEN_CEC_Controler_APP_*' in get('updata_mcu_fireware'))
commands=[n.value for n in ast.walk(mcu) if isinstance(n,ast.Constant) and isinstance(n.value,str) and n.value.startswith('i2c')]
check('Exact MCU version command reads address0x15 register0x11',commands==['i2cget -y -f 4 0x15 0x11'])
check('New package marker copies MCU candidate to /boot','atomic_write_from(mcu_path, dst)' in get('updata_mcu_fireware'))
check('N76 identity/output registers absent from this specific function',all(s not in get('updata_mcu_fireware').lower() for s in ('0x31','0x26','0x5a','0x5b','0xe0','0xe1','0xa0','0xb1')))
outer=next(n for n in main.body if isinstance(n,ast.If) and ast.unparse(n.test)=='isRK3588()' and any(isinstance(c,ast.Call) and isinstance(c.func,ast.Name) and c.func.id=='update_spl_for_dh4300plus_rk3588c' for c in ast.walk(n)))
arm_calls=[n for stmt in outer.body for n in ast.walk(stmt) if isinstance(n,ast.Call) and isinstance(n.func,ast.Name) and n.func.id=='updata_mcu_fireware']
other_calls=[n for stmt in outer.orelse for n in ast.walk(stmt) if isinstance(n,ast.Call) and isinstance(n.func,ast.Name) and n.func.id=='updata_mcu_fireware']
check('MCU updater call gated by rk3588',len(arm_calls)==1 and not other_calls)
check('rk3588 gate selected by explicit arch_spu argument',ast.unparse(names['isRK3588'].body[0])=="return arch_spu == 'rk3588'")
manifest=json.loads((OUT/'ugpt-analysis.json').read_text())
plugins=[r['name'] for r in manifest['members'] if r['name'].startswith('./plugin/')]
check('Actual archive contains only echo and upgrade plugins',sorted(plugins)==['./plugin/echo_tool.py','./plugin/upgrade_tool.py'])
def row(n):
    calls=[]
    for c in ast.walk(n):
        if isinstance(c,ast.Call): calls.append({'line':c.lineno,'function':ast.unparse(c.func)})
    return {'name':n.name,'line':n.lineno,'end_line':n.end_lineno,'calls':calls}
trace={'input':path.relative_to(ROOT).as_posix(),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
       'functions':[row(names[n]) for n in ('isRK3588','updata_mcu_fireware','update_kernel','update_kernel_rk')],
       'plugin_main':row(main),'updater_gate':{'line':outer.lineno,'condition':ast.unparse(outer.test),'mcu_call_line':arm_calls[0].lineno},
       'checks':checks,'firmware_evidence':['Original ugpt is a Makeself shell archive, gzip tar validated against embedded MD5;10 members extracted without execution',
           'Tool.main dispatches the only MCU-specific updater from explicit arch_spu=rk3588; AMD64 follows vmlinuz/initrd branch',
           'That updater names UGREEN_CEC_Controler_APP_* and reads bus4/address0x15/register0x11, then conditionally stages image under/boot',
           'No I2C flash transaction is implemented inside updata_mcu_fireware itself; staging does not prove later consumer behavior'],
       'inference':['This concrete upgrader branch is a different interface from DH2600/DXP480T N76 at0x31/0x26; it cannot define their output/PWM/timing readback'],
       'unknown':['Other untraced library or firmware payload paths are not excluded','N76 application firmware/output/PWM/timing readback still unconfirmed'],
       'performed_hardware_io':False,'executed_vendor_code':False,'support_states_changed':False}
(OUT/'report.json').write_text(json.dumps(trace,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
lines=source.splitlines(); excerpt=[]
for a,b in ((1,74),(674,722),(1310,1341),(1362,1402)):
    excerpt.extend(f'{i+1}: {lines[i]}' for i in range(a-1,b))
    excerpt.append('')
(OUT/'ugpt-branch-evidence.txt').write_text('\n'.join(excerpt)+'\n',encoding='utf-8')
print(json.dumps({'assertions':len(checks),'gate':trace['updater_gate'],'commands':commands,'classification':trace['inference']},ensure_ascii=False))
