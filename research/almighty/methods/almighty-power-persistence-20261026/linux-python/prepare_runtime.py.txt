"""Download a bounded, SHA256-index-verified Python runtime; never install it."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib
import json
import lzma
import urllib.request

ROOT = Path(__file__).resolve().parent
REPO = ROOT.parents[3]
INPUT = REPO / "firmware-inbox/input/python-qemu-20261026"
BASE = "https://deb.debian.org/debian/"
DIST = BASE + "dists/trixie/"
MAX_TOTAL = 32 * 1024 * 1024
PACKAGES = ["python3.13-minimal", "libpython3.13-minimal", "libpython3.13-stdlib",
            "libexpat1", "zlib1g", "libssl3t64", "libbz2-1.0", "liblzma5",
            "libffi8", "libsqlite3-0", "libmpdec4", "libreadline8t64", "libtinfo6",
            "libncursesw6", "libdb5.3t64", "libuuid1", "libzstd1"]

def digest(data):
    return hashlib.sha256(data).hexdigest()

def fetch(url, name, limit):
    path = INPUT / name
    if path.exists():
        data = path.read_bytes()
    else:
        with urllib.request.urlopen(url, timeout=60) as response:
            data = response.read(limit + 1)
        if len(data) > limit:
            raise RuntimeError("bounded download exceeded: " + name)
        path.write_bytes(data)
    if len(data) > limit:
        raise RuntimeError("cached input exceeded bound: " + name)
    return data

def main():
    INPUT.mkdir(parents=True, exist_ok=True)
    release_url = DIST + "InRelease"
    release = fetch(release_url, "debian-trixie-InRelease", 512 * 1024)
    lines = release.decode().splitlines()
    in_sha = False
    expected = None
    index_name = "main/binary-amd64/Packages.xz"
    for line in lines:
        if line == "SHA256:":
            in_sha = True
            continue
        if in_sha and not line.startswith(" "):
            break
        if in_sha:
            parts = line.split()
            if len(parts) == 3 and parts[2] == index_name:
                expected = (parts[0], int(parts[1]))
                break
    if expected is None:
        raise RuntimeError("official Release has no SHA256 index entry")
    index_url = DIST + index_name
    index = fetch(index_url, "debian-trixie-main-amd64-Packages.xz", 16 * 1024 * 1024)
    if (digest(index), len(index)) != expected:
        raise RuntimeError("official index failed Release SHA256/size check")
    records = {}
    for paragraph in lzma.decompress(index).decode().split("\n\n"):
        record = {}
        for line in paragraph.splitlines():
            if line and not line.startswith(" ") and ": " in line:
                key, value = line.split(": ", 1)
                record[key] = value
        if record.get("Package") in PACKAGES and record.get("Architecture") in ("amd64", "all"):
            records[record["Package"]] = record
    missing = set(PACKAGES) - set(records)
    if missing:
        raise RuntimeError("packages absent from official index: " + repr(missing))
    size = len(release) + len(index) + sum(int(records[name]["Size"]) for name in PACKAGES)
    if size > MAX_TOTAL:
        raise RuntimeError("total download budget exceeded: " + str(size))
    results = []
    for name in PACKAGES:
        record = records[name]
        filename = record["Filename"].split("/")[-1]
        url = BASE + record["Filename"]
        data = fetch(url, filename, min(int(record["Size"]), 5 * 1024 * 1024))
        if len(data) != int(record["Size"]) or digest(data) != record["SHA256"]:
            raise RuntimeError("official package SHA256/size mismatch: " + name)
        results.append({"package": name, "version": record["Version"], "url": url,
                        "file": str(INPUT / filename), "size": len(data),
                        "sha256": digest(data), "index_sha256_verified": True})
        print(name, record["Version"], len(data), flush=True)
    manifest = {"created_utc": datetime.now(timezone.utc).isoformat(),
                "source": "Debian official HTTPS archive, trixie/main/amd64",
                "signature_verification": "InRelease retained; OpenPGP signature not validated in this host-only preparation",
                "release": {"url": release_url, "file": str(INPUT / "debian-trixie-InRelease"),
                            "sha256": digest(release), "size": len(release)},
                "index": {"url": index_url, "file": str(INPUT / "debian-trixie-main-amd64-Packages.xz"),
                          "sha256": digest(index), "size": len(index), "release_sha256_verified": True},
                "packages": results, "total_bytes": size,
                "host_installation": False}
    (ROOT / "runtime-manifest.json").write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("TOTAL", size, "PACKAGE_BYTES", sum(item["size"] for item in results))

if __name__ == "__main__":
    main()
